	is_triggered_only = yes

	option = {
		name = "gfm_hfm_genocide.2.a"
		ai_chance = {
			factor = 100
		}
		if = {
			limit = { has_country_flag = gfm_hfm_target_culture }
			every_owned = {
				limit = {
					any_pop = {
						NOT = { is_primary_culture = yes }
						NOT = { is_accepted_culture = yes }

					}
				}
				every_pop = {
					limit = {
						NOT = { is_primary_culture = yes }
						NOT = { is_accepted_culture = yes }
					}
					reduce_pop = 0.80

					consciousness = 2
				}
				add_province_modifier = {
					name = gfm_hfm_recent_genocide
					duration = 3650
				}
			}
		}
		if = {
			limit = { has_country_flag = gfm_hfm_target_religion }
			every_owned = {
				limit = {
					any_pop = {
						is_state_religion = no
						size = 1000
					}
				}
				every_pop = {
					limit = {
						is_state_religion = no
					}
					reduce_pop = 0.75
					militancy = 3
					consciousness = 2
				}
				add_province_modifier = {
					name = gfm_hfm_recent_genocide
					duration = 2920
				}
			}
		}
		every_country = {
			limit = {
				is_civilized = yes
			}
			relation = {
				who = THIS
				value = -50
			}
		}

		consciousness = 1
		prestige = -15
		badboy = 10
		country_event = 4436
		clr_country_flag = gfm_hfm_target_culture
